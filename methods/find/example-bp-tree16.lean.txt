-- FIND example (local build, not a check): a 25-state branching program for tree evaluation on 16 bits, all 65,536 inputs, about 31 s CPU.
-- Write the function with direct bit reads (x 0, x 7, ...), not by building a list: the list form was over 95 s CPU and still running.
import PEqualNP.Library.Helpers.ReplayBP
open PEqualNP.BranchingProgram PEqualNP.Library.Helpers.ReplayBP
theorem find_pebble16_col_b : HasBPSize 16 (fun x : Fin 16 → Bool => let a := (if x 0 then (if x 1 then x 7 else x 6) else (if x 1 then x 5 else x 4)); let b := (if x 2 then (if x 3 then x 11 else x 10) else (if x 3 then x 9 else x 8)); if a then (if b then x 15 else x 14) else (if b then x 13 else x 12)) 25 :=
  hasBPSize_of_colReplay [⟨0, Next.state 1, Next.state 2⟩, ⟨1, Next.state 3, Next.state 4⟩, ⟨1, Next.state 5, Next.state 6⟩, ⟨4, Next.state 7, Next.state 16⟩, ⟨5, Next.state 7, Next.state 16⟩, ⟨6, Next.state 7, Next.state 16⟩, ⟨7, Next.state 7, Next.state 16⟩, ⟨2, Next.state 8, Next.state 9⟩, ⟨3, Next.state 10, Next.state 11⟩, ⟨3, Next.state 12, Next.state 13⟩, ⟨8, Next.state 14, Next.state 15⟩, ⟨9, Next.state 14, Next.state 15⟩, ⟨10, Next.state 14, Next.state 15⟩, ⟨11, Next.state 14, Next.state 15⟩, ⟨12, Next.out false, Next.out true⟩, ⟨13, Next.out false, Next.out true⟩, ⟨2, Next.state 17, Next.state 18⟩, ⟨3, Next.state 19, Next.state 20⟩, ⟨3, Next.state 21, Next.state 22⟩, ⟨8, Next.state 23, Next.state 24⟩, ⟨9, Next.state 23, Next.state 24⟩, ⟨10, Next.state 23, Next.state 24⟩, ⟨11, Next.state 23, Next.state 24⟩, ⟨14, Next.out false, Next.out true⟩, ⟨15, Next.out false, Next.out true⟩] (by decide +kernel)
